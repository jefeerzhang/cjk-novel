% ------------------------------------------------------------
%  排版网格
%
%  BodySingleLeading 是补偿值，不是行距本身。
%  \setstretch{1.3} 会把 \fontsize 的基线再乘一遍 baselinestretch，
%  所以页面上真正生效的行距 = BodySingleLeading x 1.3 = 22.5pt。
%  要改行距：先定目标行距 L，再写 L/1.3。
%  例：想让行距 24pt，就写 \newcommand{\BodySingleLeading}{24/1.3}
%
%  22.5pt 这个值本身是版面网格定的，改它之前请先重新量版。
% ------------------------------------------------------------
\newcommand{\BodySingleLeading}{17.3076923}

% \StoryParagraph{<段后间距pt>}{正文}
%   段后间距 0 表示紧接下一段，3 表示多留一点气口。
%   正文 9.5pt，行距 22.5pt。
\newcommand{\StoryParagraph}[2]{%
  \begingroup
  \rmfamily
  \setstretch{1.3}%
  \fontsize{9.5}{\BodySingleLeading}\selectfont
  #2\par
  \vspace{#1pt}
  \endgroup
}

% 三个居中的短横，用来断场景。
\newcommand{\NarrativeBreak}{%
  \par\penalty10000
  \vspace{37.4pt}
  \noindent\hbox to \textwidth{\hfil\SeparatorFont\setstretch{1.3}\fontsize{9.5}{\BodySingleLeading}\selectfont\char"002D\kern0.12em\char"002D\kern0.12em\char"002D\hfil}\par
  \vspace{31.4pt}
  \penalty10000
}

% 空白停顿：不给任何符号，只留一段空。
\newcommand{\UnprintedPause}{%
  \par\penalty10000\vspace{22pt}\penalty10000
}
